# TEST 02 — BH1750 (Освещённость)

**Датчик:** BH1750FVI  
**Интерфейс:** I2C — SDA=GPIO21, SCL=GPIO22, адрес 0x23  
**Питание:** 3.3V  
**Библиотека:** BH1750 (by Christopher Laws)

**Что проверяет:**
- Инициализация BH1750
- Чтение освещённости в люксах (0–65535 lux)
- Определение уровня освещённости (тёмно/норма/ярко)

> Адрес 0x23 — когда пин ADDR подключён к GND. При ADDR → VCC адрес = 0x5C.

---

```cpp
/*
 * NEXIS TEST — BH1750
 * Освещённость в люксах
 * I2C: SDA=21, SCL=22, адрес 0x23
 * Питание: 3.3V
 * Пин ADDR → GND (адрес 0x23)
 *
 * Библиотека: BH1750 by Christopher Laws
 */

#include <Wire.h>
#include <BH1750.h>

#define I2C_SDA      21
#define I2C_SCL      22
#define BH1750_ADDR  0x23  // ADDR→GND=0x23, ADDR→VCC=0x5C

BH1750 lightMeter;

void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println("\n=== NEXIS TEST: BH1750 ===");

  Wire.begin(I2C_SDA, I2C_SCL);

  if (!lightMeter.begin(BH1750::CONTINUOUS_HIGH_RES_MODE, BH1750_ADDR)) {
    Serial.println("[ERR] BH1750 не найден!");
    Serial.println("      Проверь: SDA=21, SCL=22, VCC=3.3V");
    Serial.println("      Пин ADDR → GND (для адреса 0x23)");
    while (true) delay(1000);
  }

  Serial.println("[OK] BH1750 инициализирован (0x23, HIGH_RES)");
  Serial.println("     Диапазон: 1 — 65535 lux");
  Serial.println("─────────────────────────");
}

void loop() {
  float lux = lightMeter.readLightLevel();

  Serial.printf("Освещённость: %.0f lux  ", lux);

  // Интерпретация уровня освещённости по ГОСТ 30494-2011
  if (lux < 0) {
    Serial.println("⚠ ошибка чтения");
  } else if (lux < 50) {
    Serial.println("[DANGER] Очень темно — риск для зрения!");
  } else if (lux < 150) {
    Serial.println("[WARN] Недостаточное освещение");
  } else if (lux < 500) {
    Serial.println("[OK] Норма для рабочего места");
  } else if (lux < 10000) {
    Serial.println("[OK] Хорошее освещение");
  } else {
    Serial.println("[OK] Яркий свет / прямой солнечный");
  }

  delay(1000);
}
```

# 📈 СТРУКТУРА ДАННЫХ

## Основные структуры:

```cpp
// Данные датчиков
struct SensorData {
    float temperature;      // °C
    float humidity;         // %
    int airQuality;         // 0-1000 (ppm CO2 эквивалент)
    int lightLevel;         // lux
    int soundLevel;         // 0-100 (относительный)
    bool motionDetected;    // true/false
    unsigned long timestamp; // millis()
};

// Состояние Pomodoro
struct PomodoroState {
    bool isActive;
    bool isWorkPhase;       // true = работа, false = отдых
    int workMinutes;        // обычно 25
    int breakMinutes;       // обычно 5
    int completedCycles;
    unsigned long startTime;
    unsigned long pauseTime;
};

// Статистика за день
struct DailyStats {
    int totalWorkMinutes;
    int totalBreakMinutes;
    int pomodoroCompleted;
    int alertsTriggered;
    float avgTemperature;
    float avgHumidity;
    int avgAirQuality;
    int sittingMinutes;
    int standingBreaks;
};

// Настройки пользователя
struct UserSettings {
    int co2Threshold;       // default: 1000
    int lightThreshold;     // default: 300
    int sittingReminder;    // minutes, default: 120
    int pomodoroWork;       // default: 25
    int pomodoroBreak;      // default: 5
    bool soundEnabled;      // default: true
    bool alertsEnabled;     // default: true
    char telegramChatId[20];
};
```

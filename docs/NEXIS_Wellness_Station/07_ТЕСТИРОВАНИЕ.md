## 2.1 Тест ESP32

### Цель: Проверить базовую работу микроконтроллера

**Код теста:**
```cpp
void test_esp32() {
    Serial.begin(115200);
    delay(1000);
    
    Serial.println("=== ESP32 Test ===");
    Serial.printf("Chip Model: %s\n", ESP.getChipModel());
    Serial.printf("CPU Freq: %d MHz\n", ESP.getCpuFreqMHz());
    Serial.printf("Flash Size: %d MB\n", ESP.getFlashChipSize() / 1024 / 1024);
    Serial.printf("Free Heap: %d KB\n", ESP.getFreeHeap() / 1024);
    
    // Проверка внутреннего LED
    pinMode(2, OUTPUT);
    for(int i = 0; i < 5; i++) {
        digitalWrite(2, HIGH);
        delay(200);
        digitalWrite(2, LOW);
        delay(200);
    }
    
    Serial.println("✅ ESP32 Test PASSED");
}
```

**Ожидаемый результат:**
```
=== ESP32 Test ===
Chip Model: ESP32-D0WDQ6
CPU Freq: 240 MHz
Flash Size: 4 MB
Free Heap: 300 KB
✅ ESP32 Test PASSED
```

## 2.2 Тест Wi-Fi

```cpp
void test_wifi() {
    Serial.println("=== WiFi Test ===");
    
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 20) {
        delay(500);
        Serial.print(".");
        attempts++;
    }
    
    if (WiFi.status() == WL_CONNECTED) {
        Serial.println();
        Serial.printf("✅ Connected to: %s\n", WIFI_SSID);
        Serial.printf("IP Address: %s\n", WiFi.localIP().toString().c_str());
        Serial.printf("Signal: %d dBm\n", WiFi.RSSI());
        Serial.println("✅ WiFi Test PASSED");
    } else {
        Serial.println("\n❌ WiFi Test FAILED");
    }
}
```

**Критерии:**
- [ ] Подключение к Wi-Fi < 10 секунд
- [ ] Сигнал > -70 dBm
- [ ] Стабильное соединение

## 2.3 Тест DHT22

### Цель: Проверить работу датчика температуры и влажности

**Код теста:**
```cpp
#include <DHT.h>

#define DHTPIN 4
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

void test_dht22() {
    Serial.println("=== DHT22 Test ===");
    
    dht.begin();
    delay(2000);  // DHT22 требует задержку
    
    for(int i = 0; i < 5; i++) {
        float temp = dht.readTemperature();
        float hum = dht.readHumidity();
        
        if (isnan(temp) || isnan(hum)) {
            Serial.println("❌ DHT22 read error!");
        } else {
            Serial.printf("Reading %d: %.1f°C, %.1f%%\n", i+1, temp, hum);
        }
        delay(2000);
    }
    
    Serial.println("✅ DHT22 Test PASSED");
}
```

**Критерии:**
- [ ] Температура в диапазоне 15-35°C (комнатная)
- [ ] Влажность в диапазоне 20-80%
- [ ] Нет NaN (not a number)
- [ ] Стабильные показания (разброс < 2°C)

## 2.9 Тест кнопок

### Цель: Проверить управление

**Код теста:**
```cpp
#define BTN_MODE 32
#define BTN_OK 33
#define BTN_CANCEL 5

void test_buttons() {
    Serial.println("=== Buttons Test ===");
    
    pinMode(BTN_MODE, INPUT_PULLUP);
    pinMode(BTN_OK, INPUT_PULLUP);
    pinMode(BTN_CANCEL, INPUT_PULLUP);
    
    Serial.println("Нажмите каждую кнопку...");
    
    bool mode_tested = false;
    bool ok_tested = false;
    bool cancel_tested = false;
    
    while (!mode_tested || !ok_tested || !cancel_tested) {
        if (digitalRead(BTN_MODE) == LOW && !mode_tested) {
            Serial.println("✅ MODE нажата!");
            mode_tested = true;
            delay(300);
        }
        if (digitalRead(BTN_OK) == LOW && !ok_tested) {
            Serial.println("✅ OK нажата!");
            ok_tested = true;
            delay(300);
        }
        if (digitalRead(BTN_CANCEL) == LOW && !cancel_tested) {
            Serial.println("✅ CANCEL нажата!");
            cancel_tested = true;
            delay(300);
        }
        delay(100);
    }
    
    Serial.println("✅ Buttons Test PASSED");
}
```

**Критерии:**
- [ ] Все 3 кнопки реагируют
- [ ] Нет ложных срабатываний
- [ ] Нет дребезга контактов

## 3.1 Тест I2C шины

### Цель: Проверить совместную работу устройств на I2C

**Код теста:**
```cpp
void test_i2c_bus() {
    Serial.println("=== I2C Bus Test ===");
    
    Wire.begin(21, 22);
    
    byte error, address;
    int nDevices = 0;
    
    for(address = 1; address < 127; address++) {
        Wire.beginTransmission(address);
        error = Wire.endTransmission();
        
        if (error == 0) {
            Serial.printf("✅ Device at 0x%02X\n", address);
            nDevices++;
        }
    }
    
    if (nDevices == 2) {  // OLED + BH1750
        Serial.println("✅ I2C Bus Test PASSED");
    } else {
        Serial.printf("❌ Expected 2 devices, found %d\n", nDevices);
    }
}
```

**Ожидаемый результат:**
```
=== I2C Bus Test ===
✅ Device at 0x23  // BH1750
✅ Device at 0x3C  // OLED
✅ I2C Bus Test PASSED
```
